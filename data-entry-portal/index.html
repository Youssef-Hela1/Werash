create table if not exists car_models (
  id bigint generated always as identity primary key,
  brand text not null,
  model_name text not null,
  start_year integer not null,
  end_year integer not null,
  image_base64 text,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

alter table car_models enable row level security;
create policy "Allow select" on car_models for select using (true);
create policy "Allow insert" on car_models for insert with check (true);
create policy "Allow update" on car_models for update using (true);
create policy "Allow delete" on car_models for delete using (true);
